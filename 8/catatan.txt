#!/bin/sh

# NOMOR 8 - DHCP PACKET CAPTURE
# Kelompok K22
# Node: tengen

INTERFACE="eth0"

if ! command -v tcpdump >/dev/null 2>&1; then
    apt-get install -y tcpdump || exit 1
fi

case "$1" in
    net2)
        FILE="/root/no8-net2.pcap"
        ;;
    net3)
        FILE="/root/no8-net3.pcap"
        ;;
    baca2)
        tcpdump -n -vvv -r /root/no8-net2.pcap
        exit
        ;;
    baca3)
        tcpdump -n -vvv -r /root/no8-net3.pcap
        exit
        ;;
    *)
        echo "Cara menggunakan:"
        echo "sh /root/no8-capture.sh net2"
        echo "sh /root/no8-capture.sh net3"
        echo "sh /root/no8-capture.sh baca2"
        echo "sh /root/no8-capture.sh baca3"
        exit 1
        ;;
esac

echo "================================"
echo " NOMOR 8 - DHCP PACKET CAPTURE"
echo " Kelompok K22"
echo "================================"
echo "Interface : $INTERFACE"
echo "File      : $FILE"
echo ""
echo "Menunggu paket DHCP..."
echo "Jalankan udhcpc pada client."
echo "Tekan CTRL+C untuk berhenti."
echo ""

tcpdump -i "$INTERFACE" -n -s 0 -U \
-w "$FILE" \
'udp port 67 or udp port 68'
