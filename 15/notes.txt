nanami=0
todo=0
gagal=0

for i in $(seq 1 100); do
    worker=$(curl -s --connect-timeout 3 \
        -H 'Host: api.jujutsu.K22.com' \
        http://127.0.0.1/ | \
        grep -oP '"worker"\s*:\s*"\K[^"]+')

    case "$worker" in
        nanami) ((nanami+=1)) ;;
        todo)   ((todo+=1)) ;;
        *)      ((gagal+=1)) ;;
    esac
done

echo "Nanami: $nanami"
echo "Todo  : $todo"
echo "Gagal : $gagal"
